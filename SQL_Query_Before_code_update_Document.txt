  `id` BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
  `employee_id` BIGINT UNSIGNED NOT NULL,
  `company_id` BIGINT UNSIGNED NOT NULL,
  `changed_by_user_id` BIGINT UNSIGNED NULL,
  `actor_type` VARCHAR(32) NULL,
  `source` VARCHAR(32) NOT NULL DEFAULT 'web_admin',
  `action` VARCHAR(16) NOT NULL DEFAULT 'update',
  `snapshot_before` JSON NULL,
  `snapshot_after` JSON NULL,
  `modules_touched` JSON NULL,
  `request_meta` JSON NULL,
  `created_at` TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (`id`),
  KEY `employee_data_change_logs_employee_id_index` (`employee_id`),
  KEY `employee_data_change_logs_company_id_index` (`company_id`),
  KEY `employee_data_change_logs_changed_by_user_id_index` (`changed_by_user_id`),
  KEY `employee_data_change_logs_company_id_created_at_index` (`company_id`, `created_at`),
  KEY `employee_data_change_logs_employee_id_created_at_index` (`employee_id`, `created_at`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

  `company_address` TEXT NULL,

  `billing_year` SMALLINT NOT NULL,

  `invoice_date` DATE NOT NULL,
  `due_date` DATE NOT NULL,
  `source` ENUM('auto','manual') NOT NULL DEFAULT 'auto',
  `status` ENUM('draft','generated','sent','paid','cancelled') NOT NULL DEFAULT 'generated',
  `auto_period_key` VARCHAR(64) NULL,
  `active_employees_count` INT NOT NULL DEFAULT 0,
  `price_per_employee` DECIMAL(12,2) NOT NULL,

CREATE TABLE IF NOT EXISTS `crm_invoice_items` (
  `id` BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
  `invoice_id` BIGINT UNSIGNED NOT NULL,
  `description` VARCHAR(500) NOT NULL,
  `quantity` INT NOT NULL DEFAULT 1,
  `unit_price` DECIMAL(12,2) NOT NULL,
  `line_total` DECIMAL(14,2) NOT NULL,
  `sort_order` INT NOT NULL DEFAULT 0,
  `created_at` TIMESTAMP NULL DEFAULT NULL,
  `updated_at` TIMESTAMP NULL DEFAULT NULL,
  PRIMARY KEY (`id`),
  KEY `fk_crm_item_invoice` (`invoice_id`),
  CONSTRAINT `fk_crm_item_invoice` FOREIGN KEY (`invoice_id`) REFERENCES `crm_invoices` (`id`) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

CREATE TABLE IF NOT EXISTS `crm_invoice_email_logs` (
  `id` BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
  `invoice_id` BIGINT UNSIGNED NOT NULL,
  `recipients` TEXT NOT NULL,
  `subject` VARCHAR(255) NULL,
  `status` ENUM('queued','sent','failed') NOT NULL DEFAULT 'queued',
  `error_message` TEXT NULL,
  `sent_by` BIGINT UNSIGNED NULL,
  `sent_at` DATETIME NULL,
  `created_at` TIMESTAMP NULL DEFAULT NULL,
  `updated_at` TIMESTAMP NULL DEFAULT NULL,
  PRIMARY KEY (`id`),
  KEY `fk_crm_email_invoice` (`invoice_id`),
  CONSTRAINT `fk_crm_email_invoice` FOREIGN KEY (`invoice_id`) REFERENCES `crm_invoices` (`id`) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;
